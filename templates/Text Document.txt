    <script>
        const meningId = localStorage.getItem("oquvchi_id");

        function almashtirTur(tur) {
            document.getElementById("haftaFilterGroup").style.display = (tur === 'oylik') ? 'none' : 'flex';
            yuklashReyting();
        }

        async function yuklashReyting() {
            const fan = document.getElementById("selFan").value;
            const tur = document.getElementById("selTur").value;
            const sinf = document.getElementById("selSinf").value;
            const oy = document.getElementById("selOy").value;
            const hafta = document.getElementById("selHafta").value;

            try {
                const url = `/api/reyting?fan=${encodeURIComponent(fan)}&bosqich_turi=${tur}&oy_nomi=${encodeURIComponent(oy)}&hafta_raqami=${hafta}&sinf=${sinf}`;
                const res = await fetch(url);
                const data = await res.json();
                const ro'yxat = data.reyting || [];

                yangilashPodium(ro'yxat);
                yangilashJadval(ro'yxat);
            } catch(e) {
                console.error(e);
            }
        }

        function yangilashPodium(list) {
            const box = document.getElementById("podiumBox");
            if (list.length === 0) {
                box.style.display = 'none';
                return;
            }
            box.style.display = 'flex';

            // 1-o'rin
            if (list[0]) {
                document.getElementById("podium1").style.display = 'block';
                document.getElementById("p1_fio").innerText = list[0].fio;
                document.getElementById("p1_meta").innerText = `${list[0].sinf}-sinf, ${list[0].maktab}`;
                document.getElementById("p1_ball").innerText = `${list[0].ball} ball`;
            } else {
                document.getElementById("podium1").style.display = 'none';
            }

            // 2-o'rin
            if (list[1]) {
                document.getElementById("podium2").style.display = 'block';
                document.getElementById("p2_fio").innerText = list[1].fio;
                document.getElementById("p2_meta").innerText = `${list[1].sinf}-sinf, ${list[1].maktab}`;
                document.getElementById("p2_ball").innerText = `${list[1].ball} ball`;
            } else {
                document.getElementById("podium2").style.display = 'none';
            }

            // 3-o'rin
            if (list[2]) {
                document.getElementById("podium3").style.display = 'block';
                document.getElementById("p3_fio").innerText = list[2].fio;
                document.getElementById("p3_meta").innerText = `${list[2].sinf}-sinf, ${list[2].maktab}`;
                document.getElementById("p3_ball").innerText = `${list[2].ball} ball`;
            } else {
                document.getElementById("podium3").style.display = 'none';
            }
        }

        function yangilashJadval(list) {
            const tbody = document.getElementById("reytingTableBody");
            if (list.length === 0) {
                tbody.innerHTML = `<tr><td colspan="6" style="text-align: center; color: #94a3b8; padding: 35px;">Ushbu davrda hali natijalar kiritilmagan</td></tr>`;
                return;
            }

            tbody.innerHTML = list.map(o => {
                let badgeClass = "rank-circle";
                if (o.orin === 1) badgeClass += " rank-1";
                else if (o.orin === 2) badgeClass += " rank-2";
                else if (o.orin === 3) badgeClass += " rank-3";

                const isMe = String(o.oquvchi_id) === String(meningId);

                return `
                    <tr class="${isMe ? 'my-row' : ''}">
                        <td><span class="${badgeClass}">${o.orin}</span></td>
                        <td>
                            <strong>${o.fio}</strong>
                            ${isMe ? '<span style="font-size: 11px; background: #2563eb; color: white; padding: 2px 6px; border-radius: 4px; margin-left: 6px;">Siz</span>' : ''}
                        </td>
                        <td>${o.sinf}-sinf</td>
                        <td>${o.maktab}</td>
                        <td>${o.viloyat || 'Buxoro'}</td>
                        <td style="text-align: right; font-weight: 800; color: #2563eb; font-size: 15px;">${o.ball} ball</td>
                    </tr>
                `;
            }).join('');
        }

        yuklashReyting();
    </script>
